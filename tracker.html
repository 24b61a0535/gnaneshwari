Load AI Brain async function loadAI() { const status = document.getElementById('ai-status'); status.innerHTML = "Initializing AI Brain... 🧠"; model = await mobilenet.load(); status.innerHTML = "✅ AI Ready! Run AI with Animation async function runAI(event) { const file = event.target.files[0]; const preview = document.getElementById('preview'); const status = document.getElementById('ai-status'); if (!file) return; const reader = new FileReader(); reader.onload = async function(e) { preview.src = e.target.result; preview.style.display = "block"; // Add "Thinking" Animation status.innerHTML = "AI is analyzing pixels... 🔍"; status.style.color = "#ffa500"; // AI Classification Delay (to show animation) setTimeout(async () => { const predictions = await model.classify(preview); const topResult = predictions[0].className; const confidence = (predictions[0].probability * 100).toFixed(1); // Success Feedback status.innerHTML = `AI Result: ${topResult} (${confidence}%)
🎉 AI Verified! +10 Bonus XP Added.`; status.style.color = "#333"; // Add Bonus XP for AI Scan updateXP(10); // Auto-select Waste Type logic if (topResult.toLowerCase().includes('bottle') || topResult.toLowerCase().includes('plastic')) { document.getElementById('wasteType').value = 'plastic'; } }, 1500); }; reader.readAsDataURL(file); } // 3. Save XP to LocalStorage function updateXP(amount) { currentXP += amount; localStorage.setItem('userXP', currentXP); console.log("Total XP: " + currentXP); } // 4.
